  <script>
    let lastGeneratedText = "";

    // 側邊欄分頁切換邏輯
    function switchTab(tabId) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
      document.getElementById('tab-' + tabId).classList.remove('hidden');

      // 更新按鈕樣式
      document.querySelectorAll('aside nav button').forEach(btn => {
        btn.classList.remove('bg-gray-900', 'text-white', 'font-medium');
        btn.classList.add('text-gray-600', 'hover:bg-gray-100');
      });
      const activeBtn = document.getElementById('nav-' + tabId);
      activeBtn.classList.add('bg-gray-900', 'text-white', 'font-medium');
      activeBtn.classList.remove('text-gray-600', 'hover:bg-gray-100');
    }

    // 下拉選單連動輸入框
    function onPresetChange() {
      const select = document.getElementById('presetSelect');
      const input = document.getElementById('promptInput');
      if (select.value) {
        input.value = select.value;
      }
    }

    // 儲存自訂項目到下拉選單
    function savePreset() {
      const input = document.getElementById('promptInput');
      const select = document.getElementById('presetSelect');
      const val = input.value.trim();
      if (!val) {
        alert('請先在輸入框輸入內容再儲存！');
        return;
      }
      // 檢查是否已存在
      let exists = false;
      for (let opt of select.options) {
        if (opt.value === val) { exists = true; break; }
      }
      if (!exists) {
        const newOpt = document.createElement('option');
        newOpt.value = val;
        newOpt.text = val;
        select.add(newOpt);
        select.value = val;
        alert('已成功儲存至常用項目！');
      } else {
        alert('此項目已經在選單中了！');
      }
    }

    // 刪除目前的自訂項目
    function deleteCustomPreset() {
      const select = document.getElementById('presetSelect');
      const input = document.getElementById('promptInput');
      if (select.selectedIndex > 0) {
        select.remove(select.selectedIndex);
        input.value = "";
        alert('已刪除該項目');
      } else {
        alert('請先從上方下拉選單選擇要刪除的項目');
      }
    }

    // 圖片生成 AI 文案
    async function generateContent() {
      const fileInput = document.getElementById('imageInput');
      const promptInput = document.getElementById('promptInput').value;
      const output = document.getElementById('resultOutput');

      if (fileInput.files.length === 0) {
        alert('請先上傳商品圖片！');
        return;
      }

      output.innerText = "AI 正在努力分析圖片與生成文案中，請稍候...";

      const file = fileInput.files[0];
      const reader = new FileReader();

      reader.onload = async function(e) {
        const base64String = e.target.result.split(',')[1];

        try {
          const response = await fetch('/api/generate', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              prompt: promptInput || "請為這款韓國女裝生成吸引人的銷售文案",
              imageBase64: base64String
            })
          });

          const data = await response.json();
          if (response.ok) {
            lastGeneratedText = data.candidates[0].content.parts[0].text;
            output.innerText = lastGeneratedText;
          } else {
            output.innerText = "生成失敗: " + (data.error || "未知錯誤");
          }
        } catch (err) {
          output.innerText = "網路連線錯誤: " + err.message;
        }
      };

      reader.readAsDataURL(file);
    }

    // Meta 一鍵發佈
    async function publishToMeta() {
      const fileInput = document.getElementById('imageInput');
      const statusDiv = document.getElementById('publishStatus');

      if (fileInput.files.length === 0) {
        alert('請先上傳商品圖片！');
        return;
      }

      const file = fileInput.files[0];
      const messageToUse = lastGeneratedText || document.getElementById('promptInput').value || "韓國女裝新品連線！";

      statusDiv.innerHTML = "<span class='text-blue-600'>⏳ 正在自動處理圖片並透過 AI 分流文案，同步發布至 FB 與 IG，請稍候...</span>";

      const reader = new FileReader();
      reader.onload = async function(e) {
        const base64String = e.target.result.split(',')[1];

        try {
          const response = await fetch('/api/publish-meta', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              message: messageToUse,
              imageBase64: base64String
            })
          });

          const result = await response.json();

          if (response.ok) {
            statusDiv.innerHTML = `
              <div class="p-4 bg-green-50 border border-green-200 rounded-lg text-green-800 space-y-2">
                <p class="font-bold">✅ 雙平台同步發布成功！</p>
                <p class="text-xs text-gray-500">自動生成圖床網址: ${result.image_url_used}</p>
                <p><strong>【FB 實際發布文案】:</strong> ${result.fb_caption_used}</p>
                <p><strong>【IG 實際發布文案】:</strong> ${result.ig_caption_used}</p>
              </div>
            `;
          } else {
            statusDiv.innerHTML = `<span class='text-red-600'>❌ 發布失敗: ${result.error || JSON.stringify(result)}</span>`;
          }
        } catch (err) {
          statusDiv.innerHTML = `<span class='text-red-600'>❌ 網路請求失敗: ${err.message}</span>`;
        }
      };

      reader.readAsDataURL(file);
    }
  </script>
